    Regression MLP: Red Wine Quality¶

    Dataset: Red Wine Quality
    Architecture: 11 → 32 → 16 → 1
    Task: Multivariable regression

    Why this architecture?¶

    Eleven physicochemical measurements feed a ReLU network with a linear output.

    Training and evaluation¶

    Fit the scaler on training data and report test MAE after converting predictions back to quality points.

    Run the complete program from the repository root:

    python src/Kafe.py docs/ejemplos/gesha-nivel-7-wine.kf

    -- Regression MLP: Red Wine Quality: multivariable regression with Red Wine Quality.
import kaggle;
import machine;
import gesha;

List[STR] columns = [
    "fixed acidity", "volatile acidity", "citric acid", "residual sugar", "chlorides",
    "free sulfur dioxide", "total sulfur dioxide", "density", "pH", "sulphates", "alcohol"
];
List[List[FLOAT]] X = kaggle.load_dataset_matrix("uciml/red-wine-quality-cortez-et-al-2009", columns, "winequality-red.csv", 240);
PARDOS df = kaggle.load_dataset("uciml/red-wine-quality-cortez-et-al-2009", "winequality-red.csv");
List[INT] quality = df.col("quality");
List[FLOAT] y = [];
for (i in range(0, len(X))):
    append(y, (float(quality[i]) - 3.0) / 5.0);
;
MACHINE split = machine.train_test_split(X, y, 0.2, 42);
List[List[FLOAT]] X_train = split[0];
List[List[FLOAT]] X_test = split[1];
List[FLOAT] y_train = split[2];
List[FLOAT] y_test = split[3];
MACHINE scaler = machine.standard_scaler();
scaler.fit(X_train);
List[List[FLOAT]] train_scaled = scaler.transform(X_train);
List[List[FLOAT]] test_scaled = scaler.transform(X_test);

GESHA h1 = gesha.create_dense(32, "relu", [11], 0.0, 42, gesha.he_normal(42));
GESHA h2 = gesha.create_dense(16, "relu", [32], 0.0, 43, gesha.he_normal(43));
GESHA output_layer = gesha.create_dense(1, "linear", [16], 0.0, 44);
GESHA model = gesha.sequential([h1, h2, output_layer]);
model.compile("adam", "mse", []);
model.set_lr(0.005);
model.fit(train_scaled, y_train, 30, 16);

FLOAT error = 0.0;
for (i in range(0, len(y_test))):
    FLOAT pred = model.predict(test_scaled[i])[0] * 5.0 + 3.0;
    FLOAT actual = y_test[i] * 5.0 + 3.0;
    FLOAT difference = pred - actual;
    if (difference < 0.0):
        difference = 0.0 - difference;
    ;
    error = error + difference;
    if (i < 5):
        show("Wine actual=" + str(actual) + " predicted=" + str(pred));
    ;
;
show("Red Wine MAE test en puntos de quality:");
show(error / len(y_test));
